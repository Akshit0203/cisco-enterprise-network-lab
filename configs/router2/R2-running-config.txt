!==============================================================================
! Device      : Router 2 (R2)
! Platform    : Cisco 2811 ISR
! IOS         : 12.4(15)T13  (C2800NM-IPBASE-M)
! Role        : Gateway for LAN B (20.0.0.0/8) and LAN D (40.0.0.0/8)
! Access      : Console (serial / PuTTY, 9600 8N1)
!------------------------------------------------------------------------------
! Lab reference configuration — CRAW Rack 1
!==============================================================================

enable
configure terminal

! --- Basic device identity ---
hostname R2
no ip domain-lookup

! --- Console & log housekeeping ---
line console 0
 logging synchronous
 exec-timeout 10 0
exit

! --- Interface: FastEthernet0/0  ->  LAN B (20.0.0.0/8) ---
interface FastEthernet0/0
 description LAN-B :: 20.0.0.0/8
 ip address 20.0.0.2 255.0.0.0
 no shutdown
exit

! --- Interface: FastEthernet0/1  ->  LAN D (40.0.0.0/8) ---
interface FastEthernet0/1
 description LAN-D :: 40.0.0.0/8
 ip address 40.0.0.1 255.0.0.0
 no shutdown
exit

! --- Static routing to R1's networks (10.0.0.0/8 and 30.0.0.0/8) ---
! Next hop is R1 across the shared segment. Adjust next-hop to match the
! inter-router link addressing used in your build.
ip route 10.0.0.0 255.0.0.0 20.0.0.2
ip route 30.0.0.0 255.0.0.0 20.0.0.2

! --- Save configuration ---
end
copy running-config startup-config

!   show ip interface brief
!   show ip route
!   ping 40.0.0.1
