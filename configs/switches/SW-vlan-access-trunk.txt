!==============================================================================
! Device      : Access/Distribution Switch (Catalyst 2960 / 3560 / 3750)
! Role        : Layer 2 segmentation — VLANs, access ports, 802.1Q trunk
!------------------------------------------------------------------------------
! Lab reference configuration — CRAW Rack 1
!==============================================================================

enable
configure terminal

hostname SW1

! --- Create VLANs ---
vlan 10
 name STAFF
exit
vlan 20
 name SERVERS
exit
vlan 99
 name MGMT
exit

! --- Access ports (end hosts / PCs) ---
interface range FastEthernet0/1 - 12
 description ACCESS :: VLAN 10 STAFF
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast             ! fast edge-port transition for hosts
exit

interface range FastEthernet0/13 - 20
 description ACCESS :: VLAN 20 SERVERS
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
exit

! --- Trunk uplink to the router / another switch (802.1Q) ---
interface GigabitEthernet0/1
 description TRUNK :: to Router (router-on-a-stick) / uplink
 switchport trunk encapsulation dot1q   ! required on 3560/3750; omit on 2960
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
exit

! --- Management SVI (in-band management on VLAN 99) ---
interface Vlan99
 description MGMT SVI
 ip address 10.0.0.3 255.0.0.0
 no shutdown
exit
ip default-gateway 10.0.0.2

! --- Save ---
end
copy running-config startup-config
